## 对称风铃

首先注意到，两个风铃平行当且仅当 $f(i)=(a_i+b_i)\bmod 2n$ 相同。假设我们需要把所有风铃转化为 $f$ 值均为 $x$ 的，那么显然 $x$ 一定不能是偶数。因为如果 $x$ 是偶数，上方和下方就会多出各一个点，但它们没办法配对。

设 $cnt$ 为出现次数最多的奇数 $f$ 值的出现次数，那么一定有最小操作次数为 $n-cnt$，因为我们可以不动这 $cnt$ 条平行线，对于 $n-cnt$ 条不满足平行的每个各一次操作扳成平行的。

为什么只用扳正好 $n-cnt$ 次呢？考虑对于一个 $(a_i,b_i)$，我们随便选一边扳到正确位置，那么正确位置哪里原有的对应彩带就会和其共用一个端点，因此我们将他再扳过去……循环往复，最后一定会形成一个环。我们便利所有的环分别进行操作就行了。

时间复杂度 $\mathcal O(n)$。注意如果用 DFS 的话，本地测试需要加开大栈空间的编译指令，否则测最大的样例会因为栈溢出导致输出不完全，留下一脸懵圈的你。

```cpp
const int kMaxN = 2e5 + 5;

int a[kMaxN], b[kMaxN], f[kMaxN], used[kMaxN], n, cnt, val = -1;
pair<int, int> nxt[kMaxN];

int calc(int x, int y) {
  return (x + y) % (2 * n);
}

void dfs(int x) {
  int tar = (val - x + 2 * n) % (2 * n);
  if (!used[nxt[x].second]) {
    cout << nxt[x].second << ' ' << nxt[x].first << ' ' << tar << '\n';
    used[nxt[x].second] = 1;
    dfs(nxt[tar].first);
  }
}

int main() {
  cin.tie(0)->sync_with_stdio(0);
  cin >> n;
  for (int i = 0; i < n; i++) {
    cin >> a[i] >> b[i];
    nxt[a[i]] = {b[i], i};
    nxt[b[i]] = {a[i], i};
    int x = calc(a[i], b[i]);
    if (x % 2 == 0) {  // 目标平行状态一定是两端和为奇数
      continue;
    }
    if (++f[x] > cnt) {
      cnt = f[x];
      val = x;
    }
  }
  if (val == -1) {
    val = 1;
  }
  for (int i = 0; i < n; i++) {
    if (calc(a[i], b[i]) == val) {
      used[i] = 1;
    }
  }
  cout << n - cnt << '\n';
  for (int i = 0; i < 2 * n; i++) {
    if (!used[nxt[i].second]) {
      dfs(i);
    }
  }
  return 0;
}
```

## 魔法石共鸣

首先，观察题目信息：将 $A$ 分成两组使得每组或值的异或和最大。由于异或和套在外面，而异或是有单调性的，因此考虑从高位到低位贪心。

对于某一位，我们想要让它的异或值为 $1$，那么分组最要满足一组只有 $0$，一组至少有一个 $1$。显然如果该位为 $1$ 的数量恰好为 $0$ 或 $n$ 的话该位只能为 $0$。

否则，由于我们是从高往低贪心的，贪低位的 $1$ 时不能因为移动导致把高位的 $1$ 又扳回为 $0$ 了，因此我们在高位分组的时候 $0$ 的和 $1$ 的严格分为两组，再低位的时候再考虑将部分 $0$ 的放到 $1$ 的那里，这样子才不会影响高位的答案。

因为在高位我们已经将所有为 $1$ 的合到一起了，并且后面只会别的数往里合并，不会将该组拆开，因此我们直接将它们视为整体，值为它们按位或的值。

```cpp
const LL kMaxN = 2e5 + 5;

LL a[kMaxN], t, n;
vector<LL> v, c;

int main() {
  cin.tie(0)->sync_with_stdio(0);
  for (cin >> t; t; t--) {
    cin >> n;
    v.clear();
    for (LL i = 1, x; i <= n; i++) {
      cin >> x;
      v.push_back(x);
    }
    LL ans = 0;
    for (LL d = 30; d >= 0; d--) {
      size_t cnt = count_if(v.begin(), v.end(), [&](LL x) {
        return x >> d & 1;
      });
      if (cnt == 0 || cnt == v.size()) {
        continue;
      }
      c.clear();
      ans |= 1 << d;
      LL res = 0;
      for (LL i : v) {
        if (i >> d & 1) {
          res |= i;
        } else {
          c.push_back(i);
        }
      }
      c.push_back(res);
      v = c;
    }
    cout << ans << '\n';
  }
  return 0;
}
```

## 能量核心共鸣

读完题面我们可以想到一个很简单粗暴的方法：动态修改用线段树实现；既然要去掉一个数使按位与最大，那么去掉的那个数一定使区间按位与的一个尽量高的位从 $0$ 扳成了 $1$。

考虑按位与的性质，我们从高位往低位枚举，如果区间按位与在该位上为 $0$ 但是该位为 $0$ 的数只有一个，那么该数就**一定**是删去最优的数。因此开 $63$ 颗线段树就可以维护了。显然时间和空间都不太行的样子。

然后我就脑残写了个看上去就不稳的方法：直接用线段树维护区间缺一个的按位与 $\max$，Push Up 时再用其与旁边的合并，然后样例都过不去！显然局部最大按位与和外侧的再进行计算不一定是最优的。我们还是得用全局贪心思想。

如何快速找到最高位满足区间内只有一个数该位为 $0$ 呢？忽然，我想到如果把之前的取 $\max$ 改成按位或合并的话，它就是可合并的了，值为区间内所有去掉一个的值按位与的或，令该值为 $\text{zrr}$、区间按位与的值为 $\text{sum}$。

对于一位，如果该位为 $0$ 的数量至少为 $2$ 个，那么哪怕去掉任意一个值的按位与该位仍然得是 $0$，因此在该位 $\text{sum}=\text{zrr}=1$；但是如果该位为 $0$ 的只出现了一次，那么删掉这个元素区间按位与在该位就会为 $1$，但没删整体为 $0$，因此该情况下 $\text{sum}=0,\text{zrr}=1$。

发现没有，我们只需要找到满足 $\text{sum}=0,\text{zrr}=1$ 的最高位，在该位持有 $0$ 的元素就是我们需要删除的目标。用 $\text{zrr}$ 与上 $\text{sum}$ 的取反，取最高位即可。

如果找到该值的具体位置？线段树上二分，如果 $\text{sum}$ 在该位也为 $0$，又该元素一定是唯一的，那么区间内必有该元素。找出来之后去掉该元素再求区间按位与即可。时间复杂度 $\mathcal O((n+q)\log n)$。

```cpp
const LL kMaxN = 1e6 + 5, kAll = (1ULL << 63) - 1;

struct Node {
  LL sum, zrr;

  friend Node operator+(const Node &a, const Node &b) {
    return Node {a.sum & b.sum, (a.zrr & b.sum) | (a.sum & b.zrr)};
  }
} tr[4 * kMaxN];

LL tag[4 * kMaxN], a[kMaxN], n, q;
bool leaf[4 * kMaxN];

void build(LL p, LL l, LL r) {
  if (l == r) {
    tr[p] = Node {a[l], kAll};
    leaf[p] = 1;
    return;
  }
  LL m = (l + r) / 2;
  build(2 * p, l, m);
  build(2 * p + 1, m + 1, r);
  tr[p] = tr[2 * p] + tr[2 * p + 1];
}

void addTag(LL p, LL d) {
  tr[p].sum &= d;
  if (!leaf[p]) {
    tr[p].zrr &= d;
  }
  tag[p] &= d;
}

void pushDown(LL p) {
  if (tag[p] != kAll) {
    addTag(2 * p, tag[p]);
    addTag(2 * p + 1, tag[p]);
    tag[p] = kAll;
  }
}

void modify(LL p, LL l, LL r, LL s, LL t, LL d) {
  if (s <= l && r <= t) {
    addTag(p, d);
    return;
  }
  pushDown(p);
  LL m = (l + r) / 2;
  if (s <= m) {
    modify(2 * p, l, m, s, t, d);
  }
  if (m < t) {
    modify(2 * p + 1, m + 1, r, s, t, d);
  }
  tr[p] = tr[2 * p] + tr[2 * p + 1];
}

void modify(LL p, LL l, LL r, LL x, LL val) {
  if (l == r) {
    tr[p] = Node {val, kAll};
    return;
  }
  pushDown(p);
  LL m = (l + r) / 2;
  if (x <= m) {
    modify(2 * p, l, m, x, val);
  } else {
    modify(2 * p + 1, m + 1, r, x, val);
  }
  tr[p] = tr[2 * p] + tr[2 * p + 1];
}

Node query(LL p, LL l, LL r, LL s, LL t) {
  if (s > t) {
    return Node {kAll, 0};
  }
  if (s <= l && r <= t) {
    return tr[p];
  }
  pushDown(p);
  LL m = (l + r) / 2;
  Node res {kAll, 0};  // Assert (res + any) = any
  if (s <= m) {
    res = res + query(2 * p, l, m, s, t);
  }
  if (m < t) {
    res = res + query(2 * p + 1, m + 1, r, s, t);
  }
  return res;
}

LL search(LL p, LL l, LL r, LL s, LL t, LL x) {
  if (l > t || r < s || !(~tr[p].sum & x)) {
    return -1;
  }
  if (l == r) {
    return l;
  }
  pushDown(p);
  LL m = (l + r) / 2, res = search(2 * p, l, m, s, t, x);
  if (res == -1) {
    res = search(2 * p + 1, m + 1, r, s, t, x);
  }
  return res;
}

int main() {
  cin.tie(0)->sync_with_stdio(0);
  cin >> n >> q;
  fill(tag + 1, tag + 4 * n + 1, kAll);
  for (LL i = 1; i <= n; i++) {
    cin >> a[i];
  }
  build(1, 1, n);
  for (LL o, x, y, z; q; q--) {
    cin >> o >> x >> y;
    if (o == 1) {
      cin >> z;
      modify(1, 1, n, x, y, z);
    } else if (o == 2) {
      modify(1, 1, n, x, y);
    } else {
      auto res = query(1, 1, n, x, y);
      LL tar = res.zrr & (~res.sum);  // 出现在 zrr 不出现在 sum 的位一定是只出现了 1 次的 0 位
      if (!tar) {
        cout << res.sum << '\n';
        continue;
      }
      tar = 1LL << __lg(tar);
      LL p = search(1, 1, n, x, y, tar);
      cout << (query(1, 1, n, x, p - 1).sum & query(1, 1, n, p + 1, y).sum) << '\n';
    }
  }
  return 0;
}
```
