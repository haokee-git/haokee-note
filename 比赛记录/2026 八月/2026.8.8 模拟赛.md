## 蜗蜗的魔法牌堆

很容易能想到一个基础策略：从小到大枚举 $i$，然后把 $i$ 元素移到 $i$ 位置上。但是它要求翻转的长度为偶数，因此 $i$ 只能每次往左走 $2^k$ 的距离。因此每次找二进制的最高 $1$ 位即可，或者令 $d$ 为 $i$ 到其正确位置的距离，若 $d$ 为奇数先交换相邻的，然后 $d$ 折半。

这样子操作次数给个超级宽松的上界是 $2n\log n$，随便过。就是模拟交换需要 $\mathcal O(n^2)$（可以使用 `std::swap_ranges`），不过本题 $n\le 10^4$ 随便搞。

```cpp
#include <iostream>
#include <algorithm>
#include <vector>
#include <ctime>

using namespace std;

const int kMaxN = 1e4 + 5;

int a[kMaxN], b[kMaxN], n;
vector<pair<int, int>> ans;

int opt(int l, int r) {
  int m = (l + r) / 2;
  swap_ranges(a + l, a + m + 1, a + m + 1);
  ans.emplace_back(l, r);
  return m;
}

int main() {
  cin.tie(0)->sync_with_stdio(0);
  cin >> n;
  for (int i = 1; i <= n; i++) {
    cin >> a[i];
  }
  for (int i = 1; i <= n; i++) {
    int p = find(a + i, a + n + 1, i) - a;
    while (p > i) {
      if ((p - i + 1) % 2) {
        p = opt(p - 1, p);
      }
      p = opt(i, p);
    }
  }
  cout << ans.size() << '\n';
  for (auto &i : ans) {
    cout << i.first << ' ' << i.second << '\n';
  }
  return 0;
}
```

## 磁力实验

首先容易用邻项交换证明每个 $a_i$ 配 $b_i$ 一定是最优的，而且写出来验证能发现总是有方案能满足 $a_i$ 能成功配上 $b_i$，因此我们只需要找到一个具体的顺序。

对于一个 $b_i$，我们要保证它能正确拿到它的 $a_i$，就得保证再它拿的时候，所有距离更近的 $a_i$ 都已经被拿完了，也就是距离更近的 $a_i$ 他们的 $b_i$ 得排在前面。它们是一个区间。

于是建有向图跑拓扑排序。边很多，但都是区间连向点，线段树优化建图整一套。求更近 $a$ 的区间采用二分。时间复杂度 $\mathcal O(n\log n)$。

```cpp
const LL kMaxN = 2e5 + 5;

LL a[kMaxN], b[kMaxN], id[kMaxN], d[4 * kMaxN], f[4 * kMaxN], n, sum;
vector<LL> e[4 * kMaxN], ans;
queue<LL> q;

void addEdge(LL u, LL v) {
  e[u].push_back(v);
  d[v]++;
}

void build(LL p, LL l, LL r) {
  if (l == r) {
    id[l] = p;
    f[p] = l;
    return;
  }
  LL m = (l + r) / 2;
  build(2 * p, l, m);
  build(2 * p + 1, m + 1, r);
  addEdge(2 * p, p);
  addEdge(2 * p + 1, p);
}

void modify(LL p, LL l, LL r, LL s, LL t, LL x) {
  if (s > t) {
    return;
  }
  if (s <= l && r <= t) {
    addEdge(p, x);
    return;
  }
  LL m = (l + r) / 2;
  if (s <= m) {
    modify(2 * p, l, m, s, t, x);
  }
  if (m < t) {
    modify(2 * p + 1, m + 1, r, s, t, x);
  }
}

int main() {
  cin.tie(0)->sync_with_stdio(0);
  cin >> n;
  for (LL i = 1; i <= n; i++) {
    cin >> a[i];
  }
  for (LL i = 1; i <= n; i++) {
    cin >> b[i];
    sum += abs(a[i] - b[i]);
  }
  build(1, 1, n);
  for (LL i = 1; i <= n; i++) {
    if (a[i] < b[i]) {
      LL p = lower_bound(a + i + 1, a + n + 1, 2 * b[i] - a[i]) - a - 1;
      modify(1, 1, n, i + 1, p, id[i]);
    } else {
      LL p = lower_bound(a + 1, a + i, 2 * b[i] - a[i]) - a;
      modify(1, 1, n, p, i - 1, id[i]);
    }
  }
  for (LL i = 1; i <= 4 * n; i++) {
    if (!d[i]) {
      q.push(i);
    }
  }
  for (; q.size(); q.pop()) {
    LL u = q.front();
    if (f[u]) {
      ans.push_back(f[u]);
    }
    for (LL v : e[u]) {
      if (!--d[v]) {
        q.push(v);
      }
    }
  }
  cout << sum << '\n';
  for (LL i : ans) {
    cout << i << ' ';
  }
  return 0;
}
```

## 符文奥秘

首先不可能直接枚举原序列，肯定是要在生成的能量序列上做章法。

将能量序列做差，相邻两项 $s_{i+1}-s_i$ 的差就等于原序列 $a_{i+k}-a_i$，这样 $a$ 会被拆分成多个环，环里相邻两个对应 $(a_i,a_{i+k})$。

显然，环的数量是 $g=\gcd(n,k)$，每个环长度为 $L=\cfrac{n}{g}$。根据题目信息，可以得到 $-1\le a_{i+k}-a_i\le 1$。
s
如果一个环不是全 $0$，那么环上一定存在至少一个差值 $-1$ 和 $1$。若差值为 $1$ 是 $0\rightarrow 1$，反之为 $1\rightarrow 0$，没有多种情况。但如果有全 $0$ 的，要么原序列上全都是 $1$，要么都是 $0$。

枚举有 $i$ 个环不是全 $0$ 的。不是全 $0$ 的原序列有 $2^L-2$ 种，因为要去掉全 $0$ 和全 $1$。对于全 $0$ 或者全 $1$ 的，有 ${} i+1$ 种方案，表示分别取 $0,1,\cdots,i$ 个原序列是全 $1$ 的。

因此，答案为

$$
\sum_{i=0}^g\binom{g}{i}(2^L-2)^i(g-i+1)
$$

令 $x=2^L-2$

$$
\sum_{i=0}^g\binom{g}{i}x^i(g-i+1)=\sum_{i=0}^g\binom{g}{i}x^i(g+1)+\sum_{i=0}^g\binom{g}{i}x^i(-i)
$$

二项式定理+吸收恒等式，得到答案为

$$
x^g+(gx)^{g-1}=(g+x+1)(x+1)^{g-1}
$$

```cpp
const LL kMod = 998244353;

LL t, n, k;

LL pow(LL a, LL b) {
  LL res = 1;
  for (; b; b /= 2) {
    b % 2 && (res = res * a % kMod);
    a = a * a % kMod;
  }
  return res;
}

int main() {
  cin.tie(0)->sync_with_stdio(0);
  for (cin >> t; t; t--) {
    cin >> n >> k;
    LL g = __gcd(n, k);
    LL zrr = n / g;
    LL x = (pow(2, zrr) - 2 + kMod) % kMod;
    cout << (g + x + 1) % kMod * pow(x + 1, g - 1) % kMod << '\n';
  }
  return 0;
}
```
