## 蜗蜗的能量水晶

显然，答案具有单调性，并且 $2nk\le 10^6$，考虑二分答案 $\mathcal O(2nk)$ 检查。更显然地，每个实验组的两堆，它们最小值的差值要尽量小，那么这两个最小值在原序列值中一定是相邻的，否则不优。

因此，先给序列排个序。判断的时候，$(a_1,a_2)$ 一定是一对最小值，特判掉之后我们需要找出剩下 $n-1$ 组的最小值，因此双指针找出上一组后面的新一组满足差值不大于二分的答案的一组，尝试将该组设为下一组的最小值。

不过不是什么组都满足条件的。如果前面有很多的无用元素没有被非最小值给选中，那么就会断档，导致中间剩下一截没被选上的。因此还需要记录前面剩下的余量，用其判断是否会发生断档。显然找靠前的最不容易断档。

```cpp
const LL kMaxN = 1e6 + 5;

LL a[kMaxN], n, k, m, ans;

bool check(LL x) {
  if (a[2] - a[1] > x) {
    return 0;
  }
  LL lst = 2, sum = 0;
  for (LL i = 2, j = 4; i <= n; i++) {
    for (; j <= m && a[j] - a[j - 1] > x; j++) { }
    if (j > m || j - lst - sum > 2 * k) {
      return 0;
    }
    sum += 2 * k - (j - lst);
    lst = j;
    j += 2;
  }
  return 1;
}

int main() {
  cin.tie(0)->sync_with_stdio(0);
  cin >> n >> k;
  m = 2 * n * k;
  for (LL i = 1; i <= m; i++) {
    cin >> a[i];
  }
  sort(a + 1, a + m + 1);
  for (LL l = 0, r = 1e9; l <= r; ) {
    LL mid = (l + r) / 2;
    if (check(mid)) {
      ans = mid;
      r = mid - 1;
    } else {
      l = mid + 1;
    }
  }
  cout << ans << '\n';
  return 0;
}
```

## 观星

为了方便本题下标采用 0-based 写法。

注意到选择两个游客在环上一个前移 $X$ 位，一个后移 $X$ 位，总有所有人的位置和在模 $m$ 意义下不变，因此可以尝试以模 $m$ 意义下位置和作为状态进行 DP。

不过，这个性质是操作合法的必要条件，而不是充分条件，因此我们还需要结合一些其它的手段使得它能转化为充要条件。

一个很 trivial 的想法是只枚举每个游客能通过 $\pm X$ 位到达的位置来做转移，但事实上这个条件结合位置总和模 $m$ 不变，一定有满足条件的操作方式可以到达。可以感性理解一下。

因此 DP 写法就水到渠成了。时间复杂度 $\mathcal O\left(nm^2\right)$ 注意虽然 $\sum n,m\le 2000$，但是单个测试点 $n,m\le 200$，也就是极限数据最多只有 $10$ 个，运算量大约 $8\times 10^7$。

```cpp
const LL kMaxN = 205;

LL a[kMaxN], h[kMaxN][kMaxN], f[kMaxN], g[kMaxN], used[kMaxN], t, n, m, x;
vector<LL> v;

int main() {
  cin.tie(0)->sync_with_stdio(0);
  for (cin >> t; t; t--) {
    cin >> n >> m >> x;
    LL sum = 0;
    for (LL i = 1; i <= n; i++) {
      cin >> a[i], a[i]--;
      sum = (sum + a[i]) % m;
    }
    for (LL i = 1; i <= n; i++) {
      for (LL j = 0; j < m; j++) {
        cin >> h[i][j];
      }
    }
    fill(f, f + m, -1e18);
    f[0] = 0;
    for (LL i = 1; i <= n; i++) {
      v.clear();
      fill(used, used + m, 0);
      for (LL j = a[i]; !used[j]; j = (j + x) % m) {
        v.push_back(j);
        used[j] = 1;
      }
      fill(g, g + m, -1e18);
      for (LL j = 0; j < m; j++) {
        if (f[j] == -1e18) {
          continue;
        }
        for (LL k : v) {
          g[(j + k) % m] = max(g[(j + k) % m], f[j] + h[i][k]);
        }
      }
      copy(g, g + m, f);
    }
    cout << f[sum] << '\n';
  }
  return 0;
}
```

## 生日聚会

首先，假设我们已经找到了每个人的正确搭档（即二分图完美匹配），那么将每对搭档缩成一个点再连边（每对点 $(u,v)$，$u$ 连上 $v$ 的所有邻点出了 $u$，$v$ 同理），生成出来的图一定是一个 DAG。

证明也很简单，如果有环的话，环上整体偏移一下，那么最佳搭档对就可以改变，但是题目中说了最佳搭档对只有一种情况，因此建出来的图是没有环的。

那为什么要把搭档对缩在一起呢？因为两搭档在答案方案中肯定是放在一起的，并且成对出现，这样答案才会最优。因此，答案就是在 DAG 上求最长路。拓扑排序+DP 解决。记录一下转移前驱还原方案。

如何找搭档？可以从度数为 $1$ 的点开始。显然一定存在至少一个度数为 $1$ 的点，如果没有就有环了。从所有度数为 $1$ 的点开始，把它的它唯一能去的点配起来，再从那个点拓展看还有没有度数为 $1$ 的点继续做，类拓扑排序。

时间复杂度 $\mathcal O(n+m)$。

```cpp
const int kMaxN = 1e6 + 5;

int d[kMaxN], p[kMaxN], n, m;
vector<int> e[kMaxN], g[kMaxN], tp;
pair<int, int> dp[kMaxN];
queue<int> q;

void dfs(int x) {
  if (dp[x].second) {
    dfs(dp[x].second);
  }
  cout << p[x] << ' ' << x << ' ';
}

int main() {
  cin.tie(0)->sync_with_stdio(0);
  cin >> n >> m;
  n *= 2;
  for (int i = 1, u, v; i <= m; i++) {
    cin >> u >> v;
    e[u].push_back(v);
    e[v].push_back(u);
    d[u]++, d[v]++;
  }
  for (int i = 1; i <= n; i++) {
    if (d[i] == 1) {
      q.push(i);
    }
  }
  for (; q.size(); q.pop()) {
    int u = q.front();
    for (int v : e[u]) {
      if (!p[v]) {
        p[u] = v, p[v] = u;
        for (int i : e[v]) {
          if (!p[i] && --d[i] == 1) {
            q.push(i);
          }
        }
      }
    }
  }
  fill(d + 1, d + n + 1, 0);
  for (int i = 1; i <= n; i++) {
    for (int j : e[p[i]]) {
      if (j != i) {
        g[i].push_back(j);
        d[j]++;
      }
    }
  }
  for (int i = 1; i <= n; i++) {
    if (!d[i]) {
      q.push(i);
    }
  }
  for (; q.size(); q.pop()) {
    int u = q.front();
    tp.push_back(u);
    for (int v : g[u]) {
      if (!--d[v]) {
        q.push(v);
      }
    }
  }
  reverse(tp.begin(), tp.end());
  for (int i : tp) {
    for (int j : g[i]) {
      dp[i] = max(dp[i], make_pair(dp[j].first, j));
    }
    dp[i].first += 2;
  }
  int p = max_element(dp + 1, dp + n + 1) - dp;
  cout << dp[p].first << '\n';
  dfs(p);
  return 0;
}
```
