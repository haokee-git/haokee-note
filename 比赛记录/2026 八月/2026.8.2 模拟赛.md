## 蜗蜗的魔法手环

我最开始把子串的长度看成 $2n$ 的了，还以为在非全同一个字符的情况下给原字符串排个序就是答案了。一个小时后才发现是 $n$，那么就还得思考一下。

首先我们发现，导致我们排序挂掉的罪魁祸首是出现最多的那个字符出现次数太多了。但是当它出现次数不超过 $n$ 时，直接排序显然是对的。

通过观察样例，可以得知显然若全是同一个字符那么也无解。如果出了出现次数最多的那个字符只有一个其它字符的话，那也显然无解，因为我们无法保证不构造出超过 $n$ 的连续段。

如果该字符出现次数正好是 $2n-2$，并且另外两个字符是一样的，那么也是无解的，哪怕将这两个字符平均分配也会有两个相同的子串。

对于其余的情况，可以先放 $n$ 个出现次数最多的字符，然后放一个其它类型的，再放剩余的出现次数最多的字符，再放完其它的字符。可以证明这样子是正确的。

int cnt[27], n, type;
string s;

int main() {
  cin.tie(0)->sync_with_stdio(0);
  cin >> s;
  n = s.size() / 2;
  for (char c : s) {
    cnt[c - 'a']++;
  }
  int num = 0, tar = 0;
  for (int i = 0; i < 26; i++) {
    if (cnt[i] > num) {
      tar = i;
      num = cnt[i];
    }
    type += cnt[i] != 0;
  }
  if (num <= n) {
    sort(s.begin(), s.end());
    cout << "YES\n" << s << '\n';
  } else if (num >= 2 * n - 1) {
    cout << "NO\n";
  } else {
    if (num == 2 * n - 2 && type == 2) {
      cout << "NO\n";
      return 0;
    }
cout << "YES\n";
    string t;
    for (char c : s) {
      if (c != 'a' + tar) {
        t += c;
      }
    }
    for (int i = 1; i <= n; i++) {
      cout << char('a' + tar);
      num--;
    }
    cout << t.back();
    t.pop_back();
    while (num--) {
      cout << char('a' + tar);
    }
    cout << t << '\n';
  }
  return 0;
}

## 蜗蜗的舒适旋律

首先我们只需要在询问区间 $[L,R]$ 当中找到一个 $k-j$ 最小的满足 $a_i\le a_j\le a_k$ 的 $(i,j,k)$，$i$ 前面 $k$ 后面都选上。考虑在中值 $a_j$ 上下功夫，那么我们只需要对于每个 $a_j$ 找出最后一个小于等于等于它的 $a_i$，最前一个大于等于它的 $a_k$，两边 $\mathcal O(n)$ 单调栈即可。

对于 $[L,R]$，我们需要在所有 $L\le i,k\le R$ 的当中找 $k-i$ 最小的那个。这是一个二维偏序的问题，排序之后树状数组即可。

- [c] 但是我把插入和询问序列放到一起了，然而数组没有开两倍，于是我就 RE 了，只有 $60$ 分。

```cpp
const LL kMaxN = 4e5 + 5;

struct {
  LL a, b, o, id, ans;
} a[kMaxN];

LL x[kMaxN], prv[kMaxN], nxt[kMaxN], tr[kMaxN], ans[kMaxN], t, n, q;
vector<LL> s;

void modify(LL x, LL y) {
  for (; x <= n; x += x & -x) {
    tr[x] = min(tr[x], y);
  }
}

LL query(LL x) {
  LL res = 1e18;
  for (; x; x -= x & -x) {
    res = min(res, tr[x]);
  }
  return res;
}

int main() {
  cin.tie(0)->sync_with_stdio(0);
  for (cin >> t; t; t--) {
    cin >> n >> q;
    for (LL i = 1; i <= n; i++) {
      cin >> x[i];
    }
    s.clear();
    for (LL i = 1; i <= n; i++) {
      for (; s.size() && x[s.back()] > x[i]; s.pop_back()) { }
      prv[i] = s.size() ? s.back() : 0;
      s.push_back(i);
    }
    s.clear();
    for (LL i = n; i >= 1; i--) {
      for (; s.size() && x[s.back()] < x[i]; s.pop_back()) { }
      nxt[i] = s.size() ? s.back() : n + 1;
      s.push_back(i);
    }
    for (LL i = 1; i <= n; i++) {
      a[i] = {prv[i], nxt[i], 0, 1, 0};
    }
    for (LL i = 1; i <= q; i++) {
      cin >> a[n + i].a >> a[n + i].b;
      a[n + i].o = 1;
      a[n + i].id = i;
      ans[i] = -1e18;
    }
    sort(a + 1, a + n + q + 1, [](auto a, auto b) {
      return a.a != b.a ? a.a > b.a : a.b != b.b ? a.b < b.b : a.o < b.o;
    });
    fill(tr + 1, tr + n + 1, 1e18);
    for (LL i = 1; i <= n + q; i++) { 
      if (a[i].o == 0) {
        modify(a[i].b, a[i].b - a[i].a - 2);
      } else {
        ans[a[i].id] = max(ans[a[i].id], a[i].b - a[i].a + 1 - query(a[i].b));
      }
    }
    for (LL i = 1; i <= q; i++) {
      cout << (ans[i] >= 3 ? ans[i] : -1) << '\n';
    }
  }
  return 0;
}
```

## 蜗蜗的魔力数字

*题外话*：这是一道原题。之前我补题时是用的是 Gemini 给的神秘妙妙做法，非矩阵的严格 $\mathcal O(n)$ 组合数学推导做法，但是现在完全不会了（当时就没有理解）。

我们可以像数位 DP 一样按位处理。首先等于 $n$ 的部分先不考虑，因为很简单，我们先只统计 $x<n$ 的 $\sum f_{\text{popcount}(x)}$。从前往后枚举 $i$，规定 $x$ 和 $n$ 的长度为 $i-1$ 的前缀相同，并且 $s_i=1$ 的话，我们可以让这一位等于 $0$ 从而接触后面位的限制，此时后面 $n-i$ 位可以随便选 $0/1$，共 $2^{n-i}$ 种方式。

但是这些方式系数是不同的，考虑预处理后缀和。从后往前做，对于每一位可以选 $0/1$，因此该位的后缀答案就是上一位的加上上一位的答案往后推进一位（即 $f_i$ 变成 $f_{i+1}$）。

直接做肯定不好做，直接使用矩阵。由于矩阵乘法有分配律，因此这样子显然是对的。

初始矩阵 

$$
A=\left(\begin{matrix}
f_1 \\
f_0
\end{matrix}\right)
$$
转移系数矩阵

$$
B=\left(\begin{matrix}
a & 1 \\
1 & 0
\end{matrix}\right)
$$
显然 $f_n=\left(B^n\times A\right)_{2,1}$。后缀和初始 $suf_{n+1}=A$，递推公式 $suf_{i}=suf_{i+1}+B\times suf_{i+1}$。

此时考虑前缀的影响。假设前缀当中有 $c$ 个 $1$，那么答案就相当于求和的每个 $f_i$ 变成 $f_{i+c}$，即直接给后缀答案乘上 $B^c$。

最好不要忘记加上 $x=n$ 的贡献。

时间复杂度 $\mathcal O(n)$，但是常数很大。

```cpp
const int kMaxN = 1e7 + 5, kMod = 1e9 + 7;

int f[kMaxN], n, a, ans;
string s;

struct Matrix {
  int a[2][2];

  Matrix() {
    a[0][0] = a[0][1] = a[1][0] = a[1][1] = 0;
  }

  int &operator()(int i, int j) {
    return a[i][j];
  }

  friend Matrix operator+(Matrix a, Matrix &b) {
    for (int i = 0; i <= 1; i++) {
      for (int j = 0; j <= 1; j++) {
        a(i, j) = (a(i, j) + b(i, j)) % kMod;
      }
    }
    return a;
  }

  friend Matrix operator*(Matrix &a, Matrix &b) {
    Matrix res;
    for (int k = 0; k <= 1; k++) {
      for (int i = 0; i <= 1; i++) {
        if (a(i, k)) {
          for (int j = 0; j <= 1; j++) {
            res(i, j) = (res(i, j) + 1LL * a(i, k) * b(k, j)) % kMod;
          }
        }
      }
    }
    return res;
  }
} suf[kMaxN], e, base, init;

void preprocess() {
  e(0, 0) = e(1, 1) = 1;
  base(0, 0) = a, base(0, 1) = base(1, 0) = 1;
  init(0, 0) = 1;
}

int main() {
  cin.tie(0)->sync_with_stdio(0);
  cin >> s;
  n = s.size();
  s = " " + s;

  cin >> a;
  preprocess();

  suf[n + 1] = init;
  for (int i = n; i >= 1; i--) {
    suf[i] = base * suf[i + 1] + suf[i + 1];
  }

  Matrix pre = e;
  for (int i = 1; i <= n; i++) {  // 前 i-1 保持不变，当 s[i]=1 时该位填 0，后面随便选
    if (s[i] == '1') {
      ans = (ans + (pre * suf[i + 1])(1, 0)) % kMod;
      pre = base * pre;
    }
  }

  // 统计等于的
  pre = init;
  for (int i = 1; i <= n; i++) {
    if (s[i] == '1') {
      pre = base * pre;
    }
  }

  cout << (ans + pre(1, 0)) % kMod << '\n';
  return 0;
}
```
