## 项目分组

首先，对于这种“互质”的题目，分解质因数肯定是一个好方法，这样子两个数互质就可以转化为没有共同的质因子。

考虑枚举质因子来处理，需要先对每个数预先进行分解之后存下来。显然，如果一个质因数被多个数包含，那么怎么分分到两个集合里都无法满足任意集合当中两两互质，因此该情况无解。如果该质因数只被一个数包含，那么没有约束。

这样就只剩下被两个数包含的情况了，那么这两个数就不能分到同一个集合当中。考虑图论建模，$n$ 个点表示这 $n$ 个数，不能分到同一个集合当中的连边，最后对所有联通块做黑白染色即可。注意染色出的 $0$ 和 $1$ 的部分要优先取大的，相同大小取最小值最小的。

```cpp
const int kMaxN = 2e6 + 5;

int f[kMaxN], c[kMaxN], p[kMaxN], mn[2], t, n, ans;
vector<int> v[kMaxN], e[kMaxN], pth[2];

void preprocess() {
  for (int i = 2; i < kMaxN; i++) {
    for (int j = i; j < kMaxN; j += i) {
      if (!p[j]) {
        p[j] = i;
      }
    }
  }
}

bool dfs(int x, int col) {
  if (c[x] != -1) {
    return c[x] == col;
  }
  c[x] = col;
  mn[col] = min(mn[col], x);
  pth[col].push_back(x);
  for (int i : e[x]) {
    if (!dfs(i, col ^ 1)) {
      return 0;
    }
  }
  return 1;
}

void solve() {
  cin >> n;
  fill(v + 1, v + kMaxN, vector<int>());
  fill(e + 1, e + n + 1, vector<int>());
  fill(c + 1, c + n + 1, -1);
  fill(f + 1, f + n + 1, 0);
  ans = 0;
  for (int i = 1, x; i <= n; i++) {
    cin >> x;
    while (x > 1) {
      v[p[x]].push_back(i);
      if (v[p[x]].size() > 2) {
        cout << "-1\n";
        return;
      }
      int j = p[x];
      for (; x % j == 0; x /= j) { }
    }
  }
  for (int i = 1; i <= 2e6; i++) {
    if (v[i].size() == 2) {
      e[v[i][0]].push_back(v[i][1]);
      e[v[i][1]].push_back(v[i][0]);
    }
  }
  for (int i = 1; i <= n; i++) {
    if (c[i] != -1) {
      continue;
    }
    mn[0] = mn[1] = 1e9;
    pth[0].clear(), pth[1].clear();
    if (!dfs(i, 0)) {
      cout << "-1\n";
      return;
    }
    if (pth[1].size() > pth[0].size() || pth[1].size() == pth[0].size() && mn[1] < mn[0]) {
      swap(pth[0], pth[1]);
    }
    for (int j : pth[0]) {
      f[j] = 1;
    }
    ans += pth[0].size();
  }
  cout << ans << '\n';
  for (int i = 1; i <= n; i++) {
    if (f[i]) {
      cout << i << ' ';
    }
  }
  cout << '\n';
}

int main() {
  cin.tie(0)->sync_with_stdio(0);
  preprocess();
  for (cin >> t; t; t--) {
    solve();
  }
  return 0;
}
```

## 信号清晰度分析

首先答案显然等于 $\sum\limits_{1\le l\le r\le n}f(l,r)$，其中 $f(l,r)$ 是 $s_{l\cdots r}$ 当中最长不下降子序列的长度。由于数只有 $0/1$，考虑枚举中间点，前面部分全部选 $0$，后面部分全部选 $1$。

先做出前缀和数组 $p$，那么 $f(l,r)$ 可以表示为（$l\sim k$ 选 $0$，$k+1\sim r$ 选 $1$）

$$
\begin{aligned}
f(l,r)&=\max_{k=l-1}^r((k-l+1)-(p_k-p_{l-1})+p_r-p_k) \\
&=-l+1-p_{l-1}+\max_{k=l-1}^r(k-2p_k)
\end{aligned}
$$

后半部分令 $P(i)=i-2p_i$，那么 $P(i)$ 可以管辖的最大区间 $[L,R]$ 的所有子区间都得取这个值。所以直接从前往后、从后往前两遍单调栈求。相同的值采用一边严格、一边不严格来处理。

时间复杂度 $\mathcal O(n)$。我赛时开了 $4$ 个 $10^7$ `long long` 数组，超了 $256~\text{MiB}$ 因此直接爆炸 $0$ 分。后来代码源给空间开大重测了，~~估计被骂了~~，我又变成 $80$ 分了。原来还要开 int128 啊，我是 ZRR。

```cpp
const LL kMaxN = 1e7 + 5;

LL p[kMaxN], l[kMaxN], r[kMaxN], t, n;
vector<LL> s;
char c;

LL f(LL i) {
  return 0 <= i && i <= n ? 2 * p[i] - i : (LL)-1e9;
}

int main() {
  cin.tie(0)->sync_with_stdio(0);
  s.reserve(1e7);
  for (cin >> t; t; t--) {
    cin >> n;
    for (LL i = 1; i <= n; i++) {
      cin >> c;
      p[i] = p[i - 1] + (c - '0');
    }
    LL ans = 0;
    for (LL i = 1; i <= n; i++) {
      ans += (-i + 1 + p[i - 1]) * (n - i + 1);
      ans += p[i] * i;
    }
    s.clear();
    for (LL i = 0; i <= n; i++) {
      for (; s.size() && f(s.back()) > f(i); s.pop_back()) { }
      l[i] = s.size() ? s.back() : -1;
      s.push_back(i);
    }
    s.clear();
    for (LL i = n; i >= 0; i--) {
      for (; s.size() && f(s.back()) >= f(i); s.pop_back()) { }
      r[i] = s.size() ? s.back() : n + 1;
      s.push_back(i);
    }
    for (LL i = 0; i <= n; i++) {
      ans -= ((i - l[i]) * (r[i] - i) - 1) * f(i);
    }
    cout << ans << '\n';
  }
  return 0;
}
```
